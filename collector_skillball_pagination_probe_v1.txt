(()=>{
const VERSION='jolly-skillball-pagination-probe-1.0';
const clean=s=>String(s||'').replace(/\s+/g,' ').trim();
const abs=u=>{try{return new URL(u||'',location.href).href}catch(e){return u||''}};

// Visible diagnostics panel first.
let old=document.getElementById('jolly-pagination-probe-panel'); if(old) old.remove();
const p=document.createElement('div');
p.id='jolly-pagination-probe-panel';
p.style='position:fixed;z-index:2147483647;left:6px;right:6px;top:6px;background:#111;color:#fff;padding:10px;border-radius:10px;font-size:13px;line-height:1.35;box-shadow:0 2px 14px #000a;max-height:46vh;overflow:auto;font-family:-apple-system,BlinkMacSystemFont,sans-serif';
p.innerHTML='<div style="display:flex;justify-content:space-between"><b>技玉 pagination probe</b><span id="jpp-status">起動中</span></div>'+
'<div style="height:7px;background:#444;border-radius:5px;margin:7px 0"><div id="jpp-bar" style="height:7px;width:3%;background:#ddd;border-radius:5px"></div></div>'+
'<div id="jpp-summary">初期化中</div><pre id="jpp-log" style="white-space:pre-wrap;margin:6px 0 0;font-size:11px;max-height:18vh;overflow:auto;background:#222;padding:6px;border-radius:6px"></pre>'+
'<div id="jpp-buttons" style="margin-top:7px"></div>';
(document.body||document.documentElement).appendChild(p);
const $=id=>document.getElementById(id), logEl=$('jpp-log'), statusEl=$('jpp-status'), barEl=$('jpp-bar'), summaryEl=$('jpp-summary'), btnEl=$('jpp-buttons');
const state={version:VERSION,started_at:new Date().toISOString(),steps:[],errors:[],result:null};
function log(s,k='INFO'){const x='['+new Date().toLocaleTimeString()+'] '+k+' '+s;state.steps.push(x);logEl.textContent+=(logEl.textContent?'\n':'')+x;logEl.scrollTop=logEl.scrollHeight}
function prog(n,s){barEl.style.width=n+'%';statusEl.textContent=s}
function dl(obj){const b=new Blob([JSON.stringify(obj,null,2)],{type:'application/json'}),a=document.createElement('a');a.href=URL.createObjectURL(b);a.download='jolly_skillball_pagination_probe.json';document.body.appendChild(a);a.click();setTimeout(()=>{URL.revokeObjectURL(a.href);a.remove()},1200)}
async function cp(obj){try{await navigator.clipboard.writeText(JSON.stringify(obj,null,2));alert('JSONをコピーしました')}catch(e){log(String(e),'ERROR')}}
function buttons(){btnEl.innerHTML='';for(const [t,f] of [['JSON保存',()=>dl(state.result||state)],['JSONコピー',()=>cp(state.result||state)],['閉じる',()=>p.remove()]]){const b=document.createElement('button');b.textContent=t;b.style='padding:8px;margin-right:6px';b.onclick=f;btnEl.appendChild(b)}}
buttons();

try{
 log('起動'); prog(10,'技玉タブ確認');
 const nav=[...document.querySelectorAll('.item_navbar')];
 const active=nav.find(x=>!String(x.className||'').includes('grayscale_50'));
 if(active?.id!=='item_navbar_skillball') throw new Error('技玉タブを手動で開いてから実行してください。現在='+clean(active?.textContent||'不明'));
 log('技玉タブ確認 OK','OK');

 prog(25,'ページ番号探索');
 const anchors=[...document.querySelectorAll('a')].map((a,i)=>({
   i,text:clean(a.textContent),href_raw:a.getAttribute('href')||'',href:abs(a.getAttribute('href')),
   id:a.id||'',class_name:a.className||'',onclick:a.getAttribute('onclick')||'',
   rel:a.getAttribute('rel')||'',data:[...a.attributes].filter(x=>x.name.startsWith('data-')).reduce((o,x)=>(o[x.name]=x.value,o),{}),
   parent_html:a.parentElement?.outerHTML.slice(0,4000)||''
 }));
 const pageAnchors=anchors.filter(x=>/^(?:\d+|NEXT|BACK|PREV|前|次)$/.test(x.text)||/page|offset|start|skill.*card|trump/i.test(x.href_raw+' '+x.id+' '+x.class_name+' '+x.onclick));
 log('全リンク '+anchors.length+' / pagination候補 '+pageAnchors.length);

 prog(45,'周辺DOM探索');
 const textNodes=[...document.querySelectorAll('body *')].filter(el=>{
   const t=clean(el.textContent);
   return t && t.length<500 && (/\b1\s+2\s+3\b/.test(t)||/\bNEXT\b/.test(t));
 }).slice(0,80).map((el,i)=>({i,tag:el.tagName,id:el.id||'',class_name:el.className||'',text:clean(el.textContent),html:el.outerHTML.slice(0,6000)}));
 log('ページ番号周辺DOM '+textNodes.length);

 prog(62,'フォーム探索');
 const forms=[...document.forms].map((f,i)=>({
   i,action_raw:f.getAttribute('action')||'',action:abs(f.getAttribute('action')),method:f.method||'',
   id:f.id||'',class_name:f.className||'',text:clean(f.textContent).slice(0,1200),
   controls:[...f.elements].map(e=>({tag:e.tagName,type:e.type||'',name:e.name||'',value:e.value||'',id:e.id||'',class_name:e.className||''})).slice(0,200),
   html:f.outerHTML.slice(0,8000)
 }));
 log('form '+forms.length);

 prog(76,'スクリプト参照探索');
 const scripts=[...document.scripts].map((s,i)=>({i,src:abs(s.getAttribute('src')),inline:(s.src?'':String(s.textContent||'').slice(0,12000))}));
 const relevantScripts=scripts.filter(x=>/page|next|skillball|trump|ItemSkillCardList|item_default/i.test(x.src+' '+x.inline)).slice(0,50);
 log('関連script候補 '+relevantScripts.length);

 prog(88,'技玉行確認');
 const rows=[...document.querySelectorAll('li.design_item_default_trump_list')].map((li,i)=>({
   i,text:clean(li.textContent),html:li.outerHTML.slice(0,6000),
   href:abs(li.querySelector('a')?.getAttribute('href'))
 }));
 log('技玉行 '+rows.length,'OK');

 state.result={
   version:VERSION,ok:true,collected_at:new Date().toISOString(),page_url:location.href,title:document.title,
   active_tab:{id:active.id,text:clean(active.textContent),class_name:active.className||''},
   counts:{all_anchors:anchors.length,page_anchor_candidates:pageAnchors.length,page_contexts:textNodes.length,forms:forms.length,relevant_scripts:relevantScripts.length,skillball_rows:rows.length},
   page_anchor_candidates:pageAnchors,page_contexts:textNodes,forms,relevant_scripts:relevantScripts,
   skillball_rows:rows,all_anchors:anchors.slice(0,800),steps:state.steps,errors:state.errors
 };
 window.__JOLLY_PAGINATION_PROBE=state.result;
 summaryEl.innerHTML='<b>完了</b><br>技玉行 '+rows.length+' / pageリンク候補 '+pageAnchors.length+' / 周辺DOM '+textNodes.length;
 prog(100,'完了');buttons();log('診断JSON作成','OK');
 try{completion({ok:true,version:VERSION,rows:rows.length,page_candidates:pageAnchors.length})}catch(e){}
}catch(e){
 state.errors.push({message:String(e?.message||e),stack:String(e?.stack||'')});
 log(String(e?.message||e),'ERROR');summaryEl.textContent='停止: '+String(e?.message||e);prog(100,'エラー');buttons();
 try{completion({ok:false,version:VERSION,error:String(e?.message||e)})}catch(_){}
}
})();
